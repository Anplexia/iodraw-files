        string leader "负责人"
        string address "地址"
        string zip "邮编"
        string tel "联系电话"
    }

        int teacher_id PK "教师编码"
        string name "姓名"
        string gender "性别"
        date birth "出生年月日"
        string ethnic "民族"
        string education "文化程度"

        decimal base_salary "基本工资"

    }

    PROJECT {
        int proj_id PK "项目编号"
        string name "项目名称"
        string source "项目来源"
        decimal funds "经费"
        date start_date "起时间"
        date end_date "讫时间"
